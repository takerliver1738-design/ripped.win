        local connection = signal:Connect(callback)

    local function make(className, properties, parent)
        local object = Instance.new(className)
        for property, value in pairs(properties or {}) do
            object[property] = value
        end
        object.Parent = parent
        return object
    end

    local function tween(object, duration, properties)
        if not object or not object.Parent then return end
        local oldTween = activeTweens[object]
        if oldTween then oldTween:Cancel() end
        local newTween = TweenService:Create(
            object,
            TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            properties
        )
        activeTweens[object] = newTween
        newTween:Play()
        return newTween
    end

    local function makeLine(parent, color, thickness, transparency, zIndex)
        return make("Frame", {
            AnchorPoint = Vector2.new(0.5, 0.5),
            BackgroundColor3 = color,
            BackgroundTransparency = transparency or 0,
            BorderSizePixel = 0,
            Size = UDim2.fromOffset(10, thickness or 1),
            ZIndex = zIndex or 2,
        }, parent)
    end

    local function setLine(line, x1, y1, x2, y2, thickness)
        local dx = x2 - x1
        local dy = y2 - y1
        line.Position = UDim2.fromOffset((x1 + x2) / 2, (y1 + y2) / 2)
        line.Size = UDim2.fromOffset(math.max(1, math.sqrt(dx * dx + dy * dy)), thickness or 1)
        line.Rotation = math.deg(math.atan2(dy, dx))
    end

    local function rebuildWeb()
        for _, line in ipairs(webLines) do line:Destroy() end
        table.clear(webLines)
        local camera = workspace.CurrentCamera
        if not camera then return end
        local viewport = camera.ViewportSize
        local cx, cy = viewport.X * 0.5, viewport.Y * 0.48
        local radius = math.min(viewport.X, viewport.Y) * 0.43

        for i = 1, 16 do
            local angle = i / 16 * math.pi * 2
            local x = cx + math.cos(angle) * radius
            local y = cy + math.sin(angle) * radius
            local line = makeLine(webFolder, C.blue, 1, 0.83, 2)
            setLine(line, cx, cy, x, y, 1)
            table.insert(webLines, line)
        end

        for ring = 1, 7 do
            local r = radius * ring / 7
            local previousX, previousY
            for i = 0, 64 do
                local angle = i / 64 * math.pi * 2
                local x = cx + math.cos(angle) * r
                local y = cy + math.sin(angle) * r
                if previousX then
                    local color = ring % 2 == 0 and C.purple or C.blue
                    local line = makeLine(webFolder, color, 1, 0.88, 2)
                    setLine(line, previousX, previousY, x, y, 1)
                    table.insert(webLines, line)
                end
                previousX, previousY = x, y
            end
        end
    end

    local crawlTime = 0
    local currentPos = Vector2.new(0.5, 0.46)
    local targetPos = currentPos
    local targetTimer = 0
    local lastViewport = Vector2.zero

    local function chooseTarget()
        local camera = workspace.CurrentCamera
        if not camera then return end
        local viewport = camera.ViewportSize
        local marginX = math.min(230, viewport.X * 0.27)
        local marginY = math.min(190, viewport.Y * 0.24)
        local minX, maxX = marginX / viewport.X, 1 - (marginX / viewport.X)
        local minY, maxY = marginY / viewport.Y, 1 - (marginY / viewport.Y)
        targetPos = Vector2.new(minX + math.random() * (maxX - minX), minY + math.random() * (maxY - minY))
    end
    chooseTarget()

    connect(RunService.RenderStepped, function(dt)
        if not alive or closing or not gui.Parent then return end
        local camera = workspace.CurrentCamera
        if camera then
            local viewport = camera.ViewportSize
            if viewport ~= lastViewport then
                lastViewport = viewport
                rebuildWeb()
                updateScale()
            end
        end

        crawlTime += dt
        targetTimer += dt
        if targetTimer >= 3.2 then
            targetTimer = 0
            chooseTarget()
        end

        local alpha = 1 - math.exp(-dt * 0.9)
        currentPos = currentPos:Lerp(targetPos, alpha)

        stage.Position = UDim2.fromScale(
            math.clamp(currentPos.X + math.sin(crawlTime * 2.1) * 0.008, 0.12, 0.88),
            math.clamp(currentPos.Y + math.sin(crawlTime * 2.8) * 0.012, 0.14, 0.86)
        )

        for _, leg in ipairs(legs) do
            local side, index = leg.side, leg.index
            local attachX = 170 + side * 30
            local attachY = 124 + (index - 2.5) * 12
            local phase = crawlTime * 7 + index * 1.6
            if side == 1 then phase += math.pi end
            local swing, lift = math.sin(phase), math.cos(phase)
            local kneeX = attachX + side * (25 + index * 5) + swing * 13
            local kneeY = attachY - 17 + (index - 2.5) * 5 + lift * 9
            local footX = attachX + side * (48 + index * 9) + swing * 22
            local footY = attachY + 12 + (index - 2.5) * 17 + math.abs(lift) * 10
            setLine(leg.upper, attachX, attachY, kneeX, kneeY, 9)
            setLine(leg.lower, kneeX, kneeY, footX, footY, 7)
        end

        local pulse = (math.sin(crawlTime * 3) + 1) * 0.5
        abdomenGlow.BackgroundTransparency = 0.12 + pulse * 0.25
        abdomenGradient.Rotation = 90 + math.sin(crawlTime) * 20
    end)
end

    local function pick(char)
        if C.aimPart == "Head" then return head(char) or hrp(char) end
        if C.aimPart == "Torso" then
            return char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or hrp(char)
        end
        local cam = workspace.CurrentCamera
        if not cam then return head(char) end
        local h, r = head(char), hrp(char)
        if h and r then
            if (h.Position - cam.CFrame.Position).Magnitude < (r.Position - cam.CFrame.Position).Magnitude then
                return h
            end
            return r
        end
        return h or r
    end
    local losParams = RaycastParams.new()
    losParams.FilterType = Enum.RaycastFilterType.Exclude
    losParams.IgnoreWater = true

    local function los(part)
        local cam = workspace.CurrentCamera
        if not cam then return true end
        local myChar = lp.Character
        losParams.FilterDescendantsInstances = myChar and { myChar } or {}
        local hit = workspace:Raycast(cam.CFrame.Position, part.Position - cam.CFrame.Position, losParams)
        if not hit or not hit.Instance then return true end
        local inst = hit.Instance
        return inst == part or inst:IsDescendantOf(part.Parent)
    end

    local visCache = {}
    local function visCached(plr, part)
        if not part then return false end
        local c = visCache[plr]
        local now = tick()
        if c and now - c.t < 0.2 then return c.v end
        local v = los(part)
        visCache[plr] = { v = v, t = now }
        return v
    end

    local function teammate(plr)
        if not C.teamCheck then return false end
        local mine = lp.Team
        return mine and plr.Team == mine
    end

    local function ignored(plr)
        for _, id in ipairs(C.ignore) do
            if plr.UserId == id then return true end
        end
        return false
    end

    local function isTarget(plr, char)
        if plr == lp then return false end
        if ignored(plr) then return false end
        if not char then return false end
        if teammate(plr) then return false end
        if C.aliveCheck and not alive(char) then return false end
        if C.ffCheck and ff(char) then return false end
        if C.hpFilter > 0 then
            local h = hum(char)
            if not h then return false end
            local pct = (h.Health / math.max(h.MaxHealth, 1)) * 100
            if pct > C.hpFilter then return false end
        end
        return true
    end

    local function predict(part, lead)
        local char = part.Parent
        if not char then return part.Position end
        local r = hrp(char)
        if not r then return part.Position end
        if C.predMode == "Off" or lead <= 0 then return part.Position end
        return part.Position + r.AssemblyLinearVelocity * lead * (C.predVel / 100)
    end

    local function fovNow()
        local f = C.fov
        if C.legit and f > C.legitFov then f = C.legitFov end
        return f
    end

    local targetList = {}

    local function byScore(a, b)
        return a.score < b.score
    end

    local function findTarget()
        local cam = workspace.CurrentCamera
        if not cam then return nil end
        local vp = cam.ViewportSize
        local cx, cy = vp.X / 2, vp.Y / 2
        local ml = UIS:GetMouseLocation()
        local f = fovNow()

        local count = 0

        for _, plr in ipairs(Players:GetPlayers()) do
            local char = plr.Character
            if isTarget(plr, char) then
                local part = pick(char)
                if part and part.Parent then
                    local dist = (part.Position - cam.CFrame.Position).Magnitude
                    if dist >= C.minDist and dist <= C.maxDist then
                        local sp, on = cam:WorldToViewportPoint(part.Position)
                        if on then
                            local sd = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                            if sd <= f then
                                local score
                                if C.priority == "Mouse" then
                                    score = (Vector2.new(sp.X, sp.Y) - Vector2.new(ml.X, ml.Y)).Magnitude
                                elseif C.priority == "Distance" then score = dist
                                elseif C.priority == "Health" then
                                    local h = hum(char)
                                    score = h and h.Health or 100
                                else
                                    score = sd
                                end

                                count = count + 1
                                local entry = targetList[count]
                                if not entry then
                                    entry = {}
                                    targetList[count] = entry
                                end
                                entry.part, entry.score = part, score
                            end
                        end
                    end
                end
            end
        end

        if count == 0 then return nil end

        for i = #targetList, count + 1, -1 do
            targetList[i] = nil
        end

        table.sort(targetList, byScore)

        for i = 1, count do
            local entry = targetList[i]
            if not C.wallCheck or los(entry.part) then return entry.part end
        end

        return nil
    end

    local sharedT, sharedTTime = nil, 0
    local function getSharedTarget(now)
        if now - sharedTTime < 0.04 and sharedT and sharedT.Parent then
            return sharedT
        end
        sharedTTime = now
        sharedT = findTarget()
        return sharedT
    end

    local aimState = { lastKey = false, toggled = false }
    local function aimHeld()
        if C.aimKeyMode == "AlwaysOn" then return true end
        local kc = Enum.KeyCode[C.aimKey]
        local held = kc and UIS:IsKeyDown(kc)
        if not held then
            local uc = Enum.UserInputType[C.aimKey]
            if uc then held = UIS:IsMouseButtonPressed(uc) end
        end
        if C.aimKeyMode == "Toggle" then
            if held and not aimState.lastKey then
                aimState.toggled = not aimState.toggled
            end
            aimState.lastKey = held
            return aimState.toggled
        end
        if C.scopedOnly and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
            return false
        end
        return held
    end

    local lastFire = 0
    local function doAim(now)
        if not C.aim then return nil end
        if not aimHeld() then return nil end
        local target = getSharedTarget(now)
        if not target then return nil end
        local cam = workspace.CurrentCamera
        if not cam then return nil end
        local pos = predict(target, C.predLead / 100)
        if C.lockMethod == "Mouse" and canMoveMouse then
            local sp = cam:WorldToViewportPoint(pos)
            local vp = cam.ViewportSize
            local dx, dy = sp.X - vp.X / 2, sp.Y - vp.Y / 2
            local sm = C.smooth / 100
            if C.legit and sm < C.legitSmooth / 100 then sm = C.legitSmooth / 100 end
            dx, dy = dx * sm, dy * sm
            if math.abs(dx) > 0.01 or math.abs(dy) > 0.01 then pcall(mousemoverel, dx, dy) end
        else
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, pos)
        end
        if C.autoFire and now - lastFire > C.autoFireDelay / 100 then
            lastFire = now
            shoot()
        end
        return target
    end

    local silentTarget = nil
    local function doSilent()
        if not C.silent then silentTarget = nil; return end
        local cam = workspace.CurrentCamera
        if not cam then return end
        local vp = cam.ViewportSize
        local cx, cy = vp.X / 2, vp.Y / 2
        local best, bd = nil, math.huge
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= lp and not teammate(plr) then
                local char = plr.Character
                if char then
                    local part = head(char) or hrp(char)
                    if part then
                        local sp, on = cam:WorldToViewportPoint(part.Position)
                        if on then
                            local d = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                            if d <= C.silentFov and d < bd then
                                bd = d
                                best = part
                            end
                        end
                    end
                end
            end
        end
        if best and math.random() * 100 <= C.silentChance then
            silentTarget = best
        else
            silentTarget = nil
        end
    end

    local function applySilent()
        if not C.silent then return end

        local cam = workspace.CurrentCamera
        if not cam then return end
        if not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then return end
        cam.CFrame = CFrame.lookAt(cam.CFrame.Position, silentTarget.Position)
    end

    local function doTrigger(now)
        if not C.trig then return end
        local kc = Enum.KeyCode[C.trigKey]
        local uc = Enum.UserInputType[C.trigKey]
        local held = (kc and UIS:IsKeyDown(kc)) or (uc and UIS:IsMouseButtonPressed(uc))
        if not held then return end
        local wait = C.trigDelay / 100 + math.random() * (C.trigRand / 100)
        if now - lastTrig < wait then return end
        local cam = workspace.CurrentCamera
        if not cam then return end
        trigFilter[1] = lp.Character
        local res = workspace:Raycast(cam.CFrame.Position, cam.CFrame.LookVector * 5000, trigParams)
        if res and res.Instance then
            local char = res.Instance.Parent
            local plr = char and Players:GetPlayerFromCharacter(char)
            if plr and plr ~= lp then
                local myTeam = lp.Team
                if not (C.trigTeam and myTeam ~= nil and plr.Team == myTeam) then
                    if not (C.trigHead and res.Instance.Name ~= "Head") then
                        lastTrig = now
                        if C.trigAuto then shoot() end
                    end
                end
            end
        end
    end

        for _, key in ipairs({ "box", "name", "dist", "hpBg", "hp", "hpText", "tracer", "dot", "look" }) do
            if handle[key] then handle[key].Visible = false end

        if handle.bonesOn then

        if not handle then return end
        for _, key in ipairs({ "box", "name", "dist", "hpBg", "hp", "hpText", "tracer", "dot", "look" }) do

    local function drawESP()
        if not drawingOk then return end
        local cam = workspace.CurrentCamera
        if not cam then return end
        if not C.esp then
            clearESP()
            return
        end

        local vp = cam.ViewportSize
        local camPos = cam.CFrame.Position

        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= lp then
                local handle = ESP.pool[plr]
                local char = plr.Character
                local humanoid = char and char:FindFirstChildOfClass("Humanoid")
                local root = char and char:FindFirstChild("HumanoidRootPart")
                local headPart = char and char:FindFirstChild("Head")

                local skip = false
                if not char or not humanoid or not root or humanoid.Health <= 0 then
                    skip = true
                end
                if not skip and not C.espTeammates and lp.Team ~= nil and plr.Team == lp.Team then
                    skip = true
                end

                local dist = 0
                if not skip then
                    dist = (root.Position - camPos).Magnitude
                    if dist > C.espMax then skip = true end
                end

                if skip then
                    hideESP(handle)
                    local hl = ESP.chams[plr]
                    if hl then hl.Enabled = false end
                else
                    if not handle then
                        handle = newESPHandle()
                        ESP.pool[plr] = handle
                    end
                    if handle then
                        local topPos = headPart and (headPart.Position + Vector3.new(0, 0.6, 0))
                            or (root.Position + Vector3.new(0, 2.5, 0))
                        local botPos = root.Position - Vector3.new(0, 3, 0)
                        local topScreen, topOn = cam:WorldToViewportPoint(topPos)
                        local botScreen, botOn = cam:WorldToViewportPoint(botPos)

                        if not topOn and not botOn then
                            hideESP(handle)
                        else
                            local height = math.max(math.abs(botScreen.Y - topScreen.Y), 6)
                            local width = math.max(height * 0.55, 4)
                            local left = topScreen.X - width / 2
                            local centerX, topY = topScreen.X, topScreen.Y

                            local visible = true
                            if C.espVisCheck then visible = visCached(plr, headPart or root) end
                            local color = visible and C.colVis or C.colHid

                                handle.dist.Visible = true

                                handle.tracer.Thickness = 1

                            end

                    end

            end
        end

    local function drawXhair()
        if not drawingOk then return end
        local cam = workspace.CurrentCamera
        if not cam then return end
        local vp = cam.ViewportSize
        local cx, cy = vp.X / 2, vp.Y / 2
        for i = 1, 4 do
            if xhair[i].Visible ~= C.xhair then xhair[i].Visible = C.xhair end
            if C.xhair then
                xhair[i].Thickness = C.xhairThick
                xhair[i].Color = C.xhairColor
                local g, len = C.xhairGap, C.xhairSize
                if i == 1 then
                    xhair[i].From = Vector2.new(cx, cy - g)
                    xhair[i].To = Vector2.new(cx, cy - g - len)
                elseif i == 2 then
                    xhair[i].From = Vector2.new(cx, cy + g)
                    xhair[i].To = Vector2.new(cx, cy + g + len)
                elseif i == 3 then

    local function movementStep()
        if dead then return end
        local char = lp.Character
        if not char or not char.Parent then
            stopFly()
            return
        end
        local h = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not h then
            stopFly()
            return
        end

        if C.speed or C.sprint then
            local desired = 16
            if C.speed then desired = C.speedVal end
            if C.sprint then desired = math.max(desired, C.sprintVal) end
            if h.WalkSpeed ~= desired then h.WalkSpeed = desired end
            lastAppliedSpeed = desired
        elseif lastAppliedSpeed ~= -1 then
            if h.WalkSpeed ~= 16 then h.WalkSpeed = 16 end
            lastAppliedSpeed = -1
        end

        if C.jump then
            if not h.UseJumpPower then h.UseJumpPower = true end
            if h.JumpPower ~= C.jumpVal then h.JumpPower = C.jumpVal end
        end

        if C.fly and root then
            if flyChar ~= char or not flyBody or not flyBody.Parent then
                stopFly()
                local bv = Instance.new("BodyVelocity")
                bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
                bv.P = 1250
                bv.Velocity = Vector3.zero
                bv.Parent = root
                flyBody = bv
                flyChar = char
            end
            h.PlatformStand = true
            local cam = workspace.CurrentCamera
            local move = Vector3.zero
            if cam then
                local forward = (UIS:IsKeyDown(Enum.KeyCode.W) and 1 or 0) - (UIS:IsKeyDown(Enum.KeyCode.S) and 1 or 0)
                local strafe = (UIS:IsKeyDown(Enum.KeyCode.D) and 1 or 0) - (UIS:IsKeyDown(Enum.KeyCode.A) and 1 or 0)
                move = cam.CFrame.LookVector * forward + cam.CFrame.RightVector * strafe
                if move.Magnitude > 0 then move = move.Unit end
            end
            local vertical = (UIS:IsKeyDown(Enum.KeyCode.Space) and 1 or 0) - (UIS:IsKeyDown(Enum.KeyCode.LeftControl) and 1 or 0)
            if flyBody then
                flyBody.Velocity = move * C.flySpeed + Vector3.new(0, vertical * C.flySpeed, 0)
            end
        elseif flyBody then
            stopFly()
            h.PlatformStand = false
        end

        if C.noclip then
            local t = tick()
            if t - lastNoclip > 0.1 then
                lastNoclip = t
                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
        end

        if C.bhop and UIS:IsKeyDown(Enum.KeyCode.Space) then
            local state = h:GetState()
            if state == Enum.HumanoidStateType.Landed
                or state == Enum.HumanoidStateType.Running
                or state == Enum.HumanoidStateType.RunningNoPhysics then
                h:ChangeState(Enum.HumanoidStateType.Jumping)
            end
        end
    end

    local function discoverMatchRefs()
        local refs = {}
        local scopes = { workspace, game:GetService("ReplicatedStorage"), lp }
        for _, scope in ipairs(scopes) do
            for _, child in ipairs(scope:GetChildren()) do
                if child:IsA("ValueBase") then
                    local n = string.lower(child.Name)
                    if n:find("match") or n:find("game") or n:find("round")
                        or n:find("state") or n:find("intermission")
                        or n:find("lobby") or n:find("queue") then
                        table.insert(refs, child)
                    end
                end
            end
        end
        Match.refs = refs
        Match.discovered = tick()
    end

    local function lobbySignals()
        for _, scope in ipairs({ lp, workspace }) do
            for _, key in ipairs({ "InLobby", "InMenu", "InIntermission", "Intermission", "Queued", "PreRound", "Spectating" }) do
                local ok, v = pcall(function() return scope:GetAttribute(key) end)
                if ok and v == true then return true end
            end
        end
        for _, key in ipairs({ "InMatch", "InGame", "Playing" }) do
            local ok, v = pcall(function() return lp:GetAttribute(key) end)
            if ok and v == false then return true end
        end
        for _, ref in ipairs(Match.refs) do
            if ref and ref.Parent then
                local n = string.lower(ref.Name)
                local v = ref.Value
                if type(v) == "boolean" then
                    if n:find("lobby") or n:find("menu") or n:find("intermission") or n:find("queue") then
                        if v then return true end
                    elseif v == false then
                        return true
                    end
                elseif type(v) == "string" then
                    local lv = string.lower(v)
                    if lv:find("lobby") or lv:find("intermission") or lv:find("waiting")
                        or lv:find("queue") or lv:find("menu") or lv:find("idle") or lv:find("starting") then
                        return true
                    end
                end
            end
        end
        local pg = lp:FindFirstChild("PlayerGui")
        if pg then
            for _, gui in ipairs(pg:GetChildren()) do
                if gui:IsA("ScreenGui") and gui.Enabled and gui.Name ~= "ripped_feed" then
                    local n = string.lower(gui.Name)
                    for _, token in ipairs(Match.lobbyTokens) do
                        if n:find(token) then return true end
                    end
                end
            end
        end
        return false
    end

    local function rageScan()
        local cam = workspace.CurrentCamera
        local queue = rage.queue
        if not cam then
            for i = #queue, 1, -1 do queue[i] = nil end
            return 0
        end
        local vp = cam.ViewportSize
        local cx, cy = vp.X / 2, vp.Y / 2
        local fov = C.rageUnlimited and math.huge or C.rageFov
        local count = 0
        for _, plr in ipairs(Players:GetPlayers()) do
            local char = plr.Character
            if rageValid(plr, char) then
                local part = ragePartFor(char)
                if part and part.Parent then
                    local dist = (part.Position - cam.CFrame.Position).Magnitude
                    if dist >= C.rageMinDist and dist <= C.rageMaxDist then
                        local sp, on = cam:WorldToViewportPoint(part.Position)
                        if on then
                            local sd = (Vector2.new(sp.X, sp.Y) - Vector2.new(cx, cy)).Magnitude
                            if sd <= fov and (not C.rageWall or los(part)) then
                                local h = hum(char)
                                local hp = h and h.Health or 100
                                local score
                                local pri = C.ragePriority
                                if pri == "Health" then
                                    score = hp
                                elseif pri == "Distance" then
                                    score = dist
                                elseif pri == "Threat" then
                                    local hpFrac = hp / math.max((h and h.MaxHealth) or 100, 1)
                                    score = sd * 0.7 + hpFrac * 120
                                else
                                    score = sd
                                end
                                count += 1
                                local e = queue[count]
                                if not e then e = {}; queue[count] = e end
                                e.plr, e.part, e.char, e.score = plr, part, char, score
                            end
                        end
                    end
                end
            end
        end
        for i = #queue, count + 1, -1 do queue[i] = nil end
        if count == 0 then return 0 end
        table.sort(queue, function(a, b) return a.score < b.score end)
        return count

            rage.lastSwitch = now
        end
        local n = rageScan()
        if n == 0 then
            rage.targetPlr = nil
            return nil
        end
        local top = rage.queue[1]
        if not top then rage.targetPlr = nil; return nil end
        if not rage.targetPlr then
            rage.targetPlr = top.plr
            rage.lastSwitch = now
        elseif top.plr ~= rage.targetPlr and (now - rage.lastSwitch) >= (C.rageSwitchDelay / 100) then
            rage.targetPlr = top.plr
            rage.lastSwitch = now
        end
        return rage.targetPlr
    end

                rage.keyState.toggled = not rage.keyState.toggled

            rage.keyState.last = held

    local function isReloading()

            if ok and v == true then return true end
        end

                if ok and v == true then return true end

    local function rageSnap(pos)
        local cam = workspace.CurrentCamera
        if not cam then return end
        if C.rageSmooth <= 0 then
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, pos)
            return
        end
        local want = pos - cam.CFrame.Position
        if want.Magnitude > 0.001 then
            local dir = cam.CFrame.LookVector:Lerp(want.Unit, C.rageSmooth / 100)
            cam.CFrame = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + dir)
        end
    end

        if not localCharAlive() then return false end
        return true
    end

    local function rageUpdate(now)
        local on = rageActive()
        if rage.active ~= on then
            rage.active = on
            if not on then
                rage.targetPlr = nil
                rage.locked = false
                if rageInd then rageInd.Visible = false end
                if C.rageHybrid then silentTarget = nil end
            end
        end

        if not plr then
            rage.locked = false
            if rageInd then rageInd.Visible = false end
            return
        end

        local char = plr.Character
        local part = char and ragePartFor(char)
        if not part or not part.Parent then
            rage.targetPlr = nil
            rage.locked = false
            if rageInd then rageInd.Visible = false end
            return
        end

        local pos = ragePredict(part, C.ragePredStrength / 100)
        rage.locked = math.random() * 100 <= C.rageHit
        if rage.locked then
            rageSnap(pos)
            if C.rageHybrid then silentTarget = part end
            if C.rageAuto and now - rage.lastFire >= 0.01 then
                if not (C.rageAutoReload and isReloading()) then
                    rage.lastFire = now
                    shoot()
                end
            end
        end

        if drawingOk and rageInd then
            if C.rageIndicator and rage.locked then
                local cam = workspace.CurrentCamera
                local vp = cam and cam.ViewportSize
                rageInd.Text = "RAGE | " .. plr.Name

                rageInd.Visible = false
            end
        end

            fovCircle.Color = C.rageCol
        end
    end
